def slug(text):

def markdown_lines(path):
    fenced = False
    for line in path.read_text(encoding='utf-8').splitlines():
        if re.match(r'^\s*(```|~~~)', line):
            fenced = not fenced
            continue
        if not fenced:

def check(root, files):
    anchors = {}
    for rel in files:
        seen, values = {}, set()
        for line in markdown_lines(root / rel):
            match = HEADING.match(line)
            if not match: continue
            base = slug(match.group(1)); index = seen.get(base, 0); seen[base] = index + 1
